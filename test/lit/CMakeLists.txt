# Locate the lit runner installed by `pip install lit`.
#
# WHY find_program with NO_DEFAULT_PATH instead of `python -m lit.main`:
#   `python -m lit.main` silently exits 0 without running any tests on lit
#   18.x, which has no __main__.py. This caused MorphizenMLIRLitTests to
#   always report "Passed" in CI despite never executing a single test.
#
# WHY NO_DEFAULT_PATH on Windows only:
#   On Windows, HINTS + NO_DEFAULT_PATH pins the search to Python's own Scripts/
#   dir so find_program can't grab an unrelated lit.exe from the MSVC toolchain
#   on PATH. On Unix there is no such conflict, and pip frequently installs the
#   lit console script outside the interpreter's own dir (e.g. /usr/local/bin or
#   ~/.local/bin), so the default PATH search must stay enabled there.
#
# WHY the pip console_scripts lit.exe is safe from the multiprocessing loop:
#   pip generates lit.exe as a compiled launcher that calls lit.main:main
#   directly. It is not a .py script, so Python's multiprocessing spawn method
#   never re-executes it in worker processes. A hand-written run-lit.py wrapper
#   would require an `if __name__ == "__main__":` guard for the same reason.
#
# The lit pip package must be installed (CI: actions/setup-python + pip install lit).
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# Derive Python's Scripts/ (Windows) and bin/ (Unix) dirs from the interpreter
# path — these are where pip installs console_scripts entry points like lit.
get_filename_component(_python_dir "${Python3_EXECUTABLE}" DIRECTORY)
if(WIN32)
    set(_lit_no_default_path NO_DEFAULT_PATH)
else()
    set(_lit_no_default_path "")
endif()
find_program(LIT_EXECUTABLE
    NAMES lit
    HINTS "${_python_dir}/Scripts" "${_python_dir}/bin" "${_python_dir}"
    ${_lit_no_default_path}
    REQUIRED
    DOC "lit test runner from pip install lit"
)

# Generate lit.site.cfg.py directly via configure_file so it always lands in
# ${CMAKE_CURRENT_BINARY_DIR} regardless of generator (single- or multi-config).
# configure_lit_site_cfg can write to per-config subdirs on MSVC, leaving the
# top-level binary dir empty and causing lit to report LIT0103.
set(LIT_SITE_CFG_IN_HEADER "# Generated by CMake — do not edit manually.")

# Expose HIPDNN_ENABLE_PLUGINS to lit as the `hip_plugins_enabled` feature.
# The Plugin/ pass tests only pass when the tools export their MLIR symbols so
# a dlopen'd plugin's registration binds to the host registry; gate them on it.
if(HIPDNN_ENABLE_PLUGINS)
    set(HIPDNN_PLUGINS_LIT_FEATURE "True")
else()
    set(HIPDNN_PLUGINS_LIT_FEATURE "False")
endif()
if(CMAKE_CONFIGURATION_TYPES)
    # Multi-config (MSVC): mark the mode as a Python format placeholder so
    # lit.site.cfg.py resolves it at runtime from --param build_mode=<cfg>.
    set(LLVM_BUILD_MODE "%(build_mode)s")
else()
    set(LLVM_BUILD_MODE "${CMAKE_BUILD_TYPE}")
endif()
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/lit.site.cfg.py.in
    ${CMAKE_CURRENT_BINARY_DIR}/lit.site.cfg.py
    @ONLY
)

# LIT_ARGS mirrors LLVM's LLVM_LIT_ARGS pattern: expose a cache variable so
# users and CI can override flags (e.g. -j4, --no-progress-bar) at configure
# time without editing this file. Default matches LLVM's own default (-sv).
# Worker count (-j) is intentionally omitted: lit defaults to os.cpu_count(),
# which is the correct behaviour on both dev machines and CI runners.
set(LIT_ARGS "-sv" CACHE STRING "Extra arguments passed to lit (e.g. -j4 --no-progress-bar)")

# Tools the suite needs at run time. hip-mlir-opt/hip-compiler are always our
# targets. The LIT support utilities (FileCheck/not/count) are EXCLUDE_FROM_ALL
# LLVM targets in a from-source (embedded) build and must be built before the
# suite runs; in a prebuilt-prefix build they are plain executables in the
# prefix (not targets), already on disk, so only depend on the ones that exist
# as targets here.
set(_lit_deps hip-mlir-opt hip-compiler)
foreach(_lit_tool FileCheck not count)
    if(TARGET ${_lit_tool})
        list(APPEND _lit_deps ${_lit_tool})
    endif()
endforeach()

# Add custom build target for manual invocation: cmake --build . --target check-hip-mlir-lit
#
# The sample plugin is a hard dependency: the Plugin/ subtree exercises
# the cross-DLL HIP_EP_PLUGINS path, so the test runner needs the DLL
# to be on disk before lit launches.
add_custom_target(check-hip-mlir-lit
    COMMAND ${LIT_EXECUTABLE} ${LIT_ARGS} --param=build_mode=$<CONFIG> .
    DEPENDS ${_lit_deps} hip_ep_sample_plugin
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    COMMENT "Running LIT tests for hip-mlir-compiler"
)

# Register with CTest (run via: ctest --test-dir <build> -C <config>).
# Pass "." as the test root (relative to WORKING_DIRECTORY = CMAKE_CURRENT_BINARY_DIR)
# rather than the absolute binary path, so lit always resolves against the working
# directory where lit.site.cfg.py lives regardless of generator (VS or Ninja).
add_test(
    NAME MorphizenMLIRLitTests
    COMMAND ${LIT_EXECUTABLE} ${LIT_ARGS} --param=build_mode=$<CONFIG> .
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
)
