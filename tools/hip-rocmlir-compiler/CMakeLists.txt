add_executable(hip-rocmlir-compiler
  hip-rocmlir-compiler.cpp
  ${CMAKE_SOURCE_DIR}/tools/problem-cache/nearest_conv_tune.cpp
)

# Same reason as LibHipCompiler: this target includes InitRocMLIRDialects.h,
# whose Triton headers require C++20.
set_target_properties(hip-rocmlir-compiler PROPERTIES
  CXX_STANDARD 20
  CXX_STANDARD_REQUIRED YES)

add_dependencies(hip-rocmlir-compiler
  HipDialectIncGen
  HipsrDialectIncGen
  OnnxDialectIncGen
  HipTypesIncGen
  HipOpsIncGen
  HipPassesIncGen
)

# rocMLIR (rocmlirTriton) is built in-tree against the same LLVM/MLIR as this
# executable (ENABLE_ROCMLIRTRITON), so its pipeline/tuning libraries are
# called in-process -- no dlopen, no serialized-IR hand-off.
#
# hip_link_rocmlir_closure adds the Triton objects once and appends the
# non-object archives after LibHipCompiler. The includes are applied directly
# because this executable includes rocMLIR headers (mlir/Dialect/Rock/*).
hip_rocmlir_apply_includes(hip-rocmlir-compiler)
target_link_libraries(hip-rocmlir-compiler PRIVATE
  HipSupport
  HipSchemasLib
  LibHipCompiler
  HipDialectIR
  HipsrDialectIR
  OnnxDialectIR
  HipDialectTransforms
  HipToTosa
  OnnxToHip
  MLIRIR
  MLIRParser
  MLIRPass
  MLIRTransforms
  MLIRFuncTransforms
  # LLVM/MLIR come from ${llvm_libs} -- no bundled second copy.
  ${llvm_libs}
)
# After LibHipCompiler so the GNU ld rescan group is the last link item.
hip_link_rocmlir_closure(hip-rocmlir-compiler)

# Follows what LibHipCompiler actually built, not BUILD_MOCK_RUNTIME: the
# benchmarking code lives there now and stubs itself out when the build does not
# allow HIP host APIs, so a tool flag derived from the mock setting alone would
# offer tuning that cannot run. No hip::host link either -- nothing here
# references a HIP symbol since the autotuner moved, which is why this binary no
# longer carries a HIP import at all.
if(HIPDNN_EP_ROCMLIR_AUTOTUNE_AVAILABLE)
  target_compile_definitions(hip-rocmlir-compiler
    PRIVATE HIP_ROCMLIR_AUTOTUNE=1)
else()
  target_compile_definitions(hip-rocmlir-compiler
    PRIVATE HIP_ROCMLIR_AUTOTUNE=0)
endif()

target_include_directories(hip-rocmlir-compiler PRIVATE
  ${CMAKE_SOURCE_DIR}/include
  ${CMAKE_BINARY_DIR}/include
  ${CMAKE_BINARY_DIR}/schemas
  ${CMAKE_SOURCE_DIR}/tools/problem-cache
)

# Embed every tools/problem-cache/lut/<arch>.fb. The compiler selects the blob
# whose gpu_arch matches the device and ignores the rest. An empty lut
# directory still builds: the registry count is 0 and every problem falls
# through to the in-process winner cache, then the tuning space.
file(GLOB _rocmlir_problem_cache_fbs CONFIGURE_DEPENDS
  "${CMAKE_SOURCE_DIR}/tools/problem-cache/lut/*.fb")
list(SORT _rocmlir_problem_cache_fbs)
set(_problem_cache_gen
  ${CMAKE_CURRENT_BINARY_DIR}/rocmlir_problem_cache_data.cpp)
set(_problem_cache_arrays "")
set(_problem_cache_ptrs "")
set(_problem_cache_sizes "")
set(_problem_cache_count 0)
foreach(_fb IN LISTS _rocmlir_problem_cache_fbs)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_fb})
  get_filename_component(_stem ${_fb} NAME_WE)
  file(READ ${_fb} _hex HEX)
  string(LENGTH "${_hex}" _len)
  math(EXPR _n "${_len} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _body "${_hex}")
  # CMake's regex has no {n}. Sixteen bytes per line stays under MSVC's
  # source-line limit (0xNN, is 5 characters).
  set(_hexbyte "0x[0-9a-f][0-9a-f],")
  set(_hexline "${_hexbyte}${_hexbyte}${_hexbyte}${_hexbyte}")
  set(_hexline "${_hexline}${_hexline}${_hexline}${_hexline}")
  string(REGEX REPLACE "(${_hexline})" "\\1\n" _body "${_body}")
  string(APPEND _problem_cache_arrays
    "static const unsigned char _blob_${_stem}[] = {\n${_body}\n};\n")
  string(APPEND _problem_cache_ptrs "  _blob_${_stem},\n")
  string(APPEND _problem_cache_sizes "  ${_n},\n")
  math(EXPR _problem_cache_count "${_problem_cache_count} + 1")
endforeach()
if(_problem_cache_count EQUAL 0)
  file(WRITE ${_problem_cache_gen}
"// Auto-generated by tools/hip-rocmlir-compiler/CMakeLists.txt; do not edit, do not commit.\n"
"#include <cstddef>\n"
"extern \"C\" const unsigned char* const kRocmlirProblemCacheBlobs[1] = {nullptr};\n"
"extern \"C\" const size_t kRocmlirProblemCacheBlobSizes[1] = {0};\n"
"extern \"C\" const size_t kRocmlirProblemCacheBlobCount = 0;\n")
else()
  file(WRITE ${_problem_cache_gen}
"// Auto-generated by tools/hip-rocmlir-compiler/CMakeLists.txt; do not edit, do not commit.\n"
"#include <cstddef>\n"
"${_problem_cache_arrays}"
"extern \"C\" const unsigned char* const kRocmlirProblemCacheBlobs[] = {\n${_problem_cache_ptrs}};\n"
"extern \"C\" const size_t kRocmlirProblemCacheBlobSizes[] = {\n${_problem_cache_sizes}};\n"
"extern \"C\" const size_t kRocmlirProblemCacheBlobCount = ${_problem_cache_count};\n")
endif()
target_sources(hip-rocmlir-compiler PRIVATE ${_problem_cache_gen})

install(TARGETS hip-rocmlir-compiler
  RUNTIME DESTINATION bin
)
